#!/usr/bin/env python3
# -*- coding: utf-8 -*-

"""
Простой скрипт учёта смен сторожа.
Данные сохраняются в файл shift_log.csv (разделитель — точка с запятой,
чтобы Excel корректно открывал русский текст).

Используются только стандартные библиотеки: csv, datetime, os.
"""

import csv
import os
from datetime import datetime

# Имя файла с журналом смен
LOG_FILE = "shift_log.csv"

# Заголовки столбцов в CSV
HEADERS = ["Дата", "Начало", "Конец", "Длительность (ч)", "Событие"]


def ensure_file_exists():
    """Создаёт CSV-файл с заголовками, если его ещё нет."""
    if not os.path.exists(LOG_FILE):
        # newline="" — обязательное условие для модуля csv,
        # иначе на Windows будут появляться пустые строки
        with open(LOG_FILE, mode="w", encoding="utf-8", newline="") as f:
            writer = csv.writer(f, delimiter=";")
            writer.writerow(HEADERS)
        print(f"Создан новый файл журнала: {LOG_FILE}")


def ask_date():
    """Спрашивает дату в формате ДД.ММ.ГГГГ и возвращает строку ISO (ГГГГ-ММ-ДД)."""
    while True:
        raw = input("Дата смены (ДД.ММ.ГГГГ): ").strip()
        try:
            # Проверяем, что дата корректная
            dt = datetime.strptime(raw, "%d.%m.%Y")
            return dt.strftime("%Y-%m-%d")
        except ValueError:
            print("  ! Неверный формат даты. Пример: 05.11.2026")


def ask_time(label):
    """Спрашивает время в формате ЧЧ:ММ и возвращает строку в том же формате."""
    while True:
        raw = input(f"{label} (ЧЧ:ММ): ").strip()
        try:
            dt = datetime.strptime(raw, "%H:%M")
            return dt.strftime("%H:%M")
        except ValueError:
            print("  ! Неверный формат времени. Пример: 08:30")


def calc_hours(start_str, end_str):
    """
    Считает длительность смены в часах.
    Если конец меньше начала — считаем, что смена перешла через полночь.
    """
    fmt = "%H:%M"
    start = datetime.strptime(start_str, fmt)
    end = datetime.strptime(end_str, fmt)
    if end < start:
        # смена через полночь: добавляем сутки
        end = end.replace(day=end.day)  # не нужно, но для наглядности
        delta = (datetime.strptime("23:59", fmt) - start) + (end - datetime.strptime("00:00", fmt))
        # проще: прибавить 24 часа
        delta = (end - start).total_seconds() / 3600 + 24
    else:
        delta = (end - start).total_seconds() / 3600
    return round(delta, 2)


def ask_event():
    """Спрашивает текст о том, что произошло на смене."""
    while True:
        text = input("Что произошло (текст): ").strip()
        if text:
            return text
        print("  ! Поле не должно быть пустым.")


def append_record(record):
    """Дописывает одну строку в CSV-файл."""
    with open(LOG_FILE, mode="a", encoding="utf-8", newline="") as f:
        writer = csv.writer(f, delimiter=";")
        writer.writerow(record)


def main():
    print("=== Учёт смен сторожа ===")
    ensure_file_exists()

    date_str = ask_date()
    start_str = ask_time("Время начала")
    end_str = ask_time("Время конца")
    hours = calc_hours(start_str, end_str)
    event = ask_event()

    record = [date_str, start_str, end_str, hours, event]
    append_record(record)

    print("\nЗапись сохранена:")
    print(f"  Дата:         {date_str}")
    print(f"  Начало:       {start_str}")
    print(f"  Конец:        {end_str}")
    print(f"  Длительность: {hours} ч")
    print(f"  Событие:      {event}")
    print(f"\nФайл: {os.path.abspath(LOG_FILE)}")


if __name__ == "__main__":
    main()

### Что делает скрипт

1. **Проверяет файл** `shift_log.csv`. Если его нет — создаёт с заголовками: `Дата;Начало;Конец;Длительность (ч);Событие`.
2. **Спрашивает дату** в формате `ДД.ММ.ГГГГ` (с проверкой корректности).
3. **Спрашивает время начала и конца** в формате `ЧЧ:ММ`.
4. **Считает длительность** в часах. Если смена переходит через полночь (например, `22:00` → `06:00`), скрипт это учитывает.
5. **Спрашивает текст события** (не пустой).
6. **Дописывает строку** в CSV и выводит итог на экран.

### Особенности

- Разделитель `;` — чтобы Excel в русской локали открывал файл без «танцев с бубном».
- `newline=""` при открытии — обязательное требование модуля `csv`, иначе появятся пустые строки.
- Кодировка `utf-8` — корректно сохраняет кириллицу.
- Все проверки — в циклах `while True`, чтобы новичок не «вылетал» из скрипта из-за опечатки.

# -*- coding: utf-8 -*-

"""
Скрипт учёта суточных смен сторожа с учётом посещений спецслужб.
Данные сохраняются в файл shift_log.csv (разделитель — точка с запятой,
чтобы Excel корректно открывал русский текст).

Используются только стандартные библиотеки: csv, datetime, os.
"""

import csv
import os
from datetime import datetime

# Имя файла с журналом смен
LOG_FILE = "shift_log.csv"

# Список спецслужб, посещения которых фиксируем.
# Для каждой будет два столбца: "Прибытие <имя>" и "Убытие <имя>".
SERVICES = ["Ветврач", "Молокоцех", "Бухгалтерия", "Электрик"]

# Заголовки столбцов в CSV.
# Сначала общие поля смены, потом для каждой службы — прибытие и убытие.
HEADERS = ["Дата", "Начало", "Конец", "Длительность (ч)", "Событие"]
for name in SERVICES:
    HEADERS.append(f"Прибытие {name}")
    HEADERS.append(f"Убытие {name}")


def ensure_file_exists():
    """Создаёт CSV-файл с заголовками, если его ещё нет."""
    if not os.path.exists(LOG_FILE):
        # newline="" — обязательное условие для модуля csv,
        # иначе на Windows будут появляться пустые строки

            return dt.strftime("%Y-%m-%d")
        except ValueError:
            print("  ! Неверный формат даты. Пример: 05.11.2026")


def ask_time(label):
    """Спрашивает время в формате ЧЧ:ММ и возвращает строку в том же формате."""
    while True:
        raw = input(f"{label} (ЧЧ:ММ): ").strip()
        try:
            dt = datetime.strptime(raw, "%H:%M")
            return dt.strftime("%H:%M")
        except ValueError:
            print("  ! Неверный формат времени. Пример: 08:30")


def ask_optional_time(label):
    """
    Спрашивает время в формате ЧЧ:ММ.
    Пустой ввод (Enter) означает «не было» — возвращаем пустую строку.
    Используется для прибытия/убытия спецслужб.
    """
    while True:
        raw = input(f"{label} (ЧЧ:ММ или Enter, если не было): ").strip()
        if raw == "":
            return ""  # пользователь пропустил — значит, не приезжал
        try:
            dt = datetime.strptime(raw, "%H:%M")
            return dt.strftime("%H:%M")
        except ValueError:
            print("  ! Неверный формат времени. Пример: 08:30 (или Enter)")


def calc_hours(start_str, end_str):
    """
    Считает длительность смены в часах.
    Смены суточные, поэтому если конец <= начала — считаем,
    что смена перешла через полночь, и прибавляем 24 часа.
    Пример: 08:00 -> 08:00 = 24 часа (следующий день).
    """
    fmt = "%H:%M"
    start = datetime.strptime(start_str, fmt)
    end = datetime.strptime(end_str, fmt)

    # Разница в часах
    diff = (end - start).total_seconds() / 3600

    # Если разница <= 0 — значит, конец наступил на следующий день
    if diff <= 0:
        diff += 24

    return round(diff, 2)

        if text:
            return text
        print("  ! Поле не должно быть пустым.")

    """
    По очереди спрашивает время прибытия и убытия для каждой спецслужбы.
    Возвращает список значений в порядке, соответствующем SERVICES:
    [приб1, уб1, приб2, уб2, ...].

    то и убытие не спрашиваем — сразу пишем пустые значения.
    """

    for name in SERVICES:
        arrival = ask_optional_time(f"Прибытие {name}")
        if arrival == "":
            # Специалист не приезжал — убытие тоже пустое, не спрашиваем
            departure = ""
            print(f"  — {name}: не было")
        else:
            departure = ask_optional_time(f"Убытие {name}")
        values.append(arrival)
        values.append(departure)
    return values


def append_record(record):
    """Дописывает одну строку в CSV-файл."""
    with open(LOG_FILE, mode="a", encoding="utf-8", newline="") as f:
        writer = csv.writer(f, delimiter=";")
        writer.writerow(record)

    print("=== Учёт суточных смен сторожа ===")
    ensure_file_exists()

    date_str = ask_date()
    start_str = ask_time("Время начала")
    end_str = ask_time("Время конца")
    hours = calc_hours(start_str, end_str)
    event = ask_event()
    service_values = ask_services()

    # Собираем полную строку: общие поля + данные по службам
    record = [date_str, start_str, end_str, hours, event] + service_values
    append_record(record)

    print("\nЗапись сохранена:")
    print(f"  Дата:         {date_str}")
    print(f"  Начало:       {start_str}")
    print(f"  Конец:        {end_str}")
    print(f"  Длительность: {hours} ч")
    print(f"  Событие:      {event}")
    for i, name in enumerate(SERVICES):
        arrival = service_values[i * 2]
        departure = service_values[i * 2 + 1]
        if arrival == "" and departure == "":
            print(f"  {name}: не было")
        else:
            print(f"  {name}: прибытие {arrival or '—'}, убытие {departure or '—'}")
    print(f"\nФайл: {os.path.abspath(LOG_FILE)}")


if __name__ == "__main__":
    main()

### Что изменилось

**1. Расчёт длительности для суточных смен.**
Функция `calc_hours` теперь использует условие `diff <= 0` → прибавить 24 часа. Это значит:
- `08:00 → 08:00` = **24.0 ч** (следующий день);
- `20:00 → 06:00` = **10.0 ч**;
- `08:00 → 20:00` = **12.0 ч**.

Знак `<=` (а не `<`) — потому что суточная смена «в то же время» это ровно 24 часа, а не 0.

**2. Учёт спецслужб.**
- Список `SERVICES` вынесен в начало — легко добавить/убрать специалиста одной строкой.
- Заголовки CSV формируются автоматически: для каждой службы два столбца — `Прибытие <имя>` и `Убытие <имя>`.
- Новая функция `ask_optional_time` принимает Enter как «не было».
- Функция `ask_services` опрашивает всех по очереди. Если на прибытии нажат Enter — убытие не спрашиваем (специалист не приезжал).

**3. Совместимость со старым файлом.**
⚠️ **Важно:** если у вас уже есть `shift_log.csv` со старой шапкой (без столбцов служб), новый скрипт допишет строки с другими столбцами — получится «каша». Варианты:
- удалить/переименовать старый файл (скрипт создаст новый с правильными заголовками);
- вручную дописать в первую строку старого файла недостающие заголовки: `Прибытие Ветврач;Убытие Ветврач;...` — но старые строки останутся короче, что не критично для чтения.

# Для каждой службы — один столбец "Имя (Приб/Уб)"
for name in SERVICES:
    HEADERS.append(f"{name} (Приб/Уб)")


def ensure_file_exists():
    """Создаёт CSV-файл с заголовками, если его ещё нет."""
    if not os.path.exists(LOG_FILE):
        # newline="" — обязательное условие для модуля csv,
        # иначе на Windows будут появляться пустые строки
        with open(LOG_FILE, mode="w", encoding="utf-8", newline="") as f:
            writer = csv.writer(f, delimiter=";")
            writer.writerow(HEADERS)
        print(f"Создан новый файл журнала: {LOG_FILE}")


def ask_date():
    """Спрашивает дату в формате ДД.ММ.ГГГГ и возвращает строку ISO (ГГГГ-ММ-ДД)."""
    while True:
        raw = input("Дата смены (ДД.ММ.ГГГГ): ").strip()
        try:
            # Проверяем, что дата корректная
            dt = datetime.strptime(raw, "%d.%m.%Y")
            return dt.strftime("%Y-%m-%d")
        except ValueError:
            print("  ! Неверный формат даты. Пример: 05.11.2026")


def ask_required_text(label):
    """Спрашивает непустой текст (обязательное поле)."""
    while True:
        text = input(f"{label}: ").strip()
        if text:
            return text
        print("  ! Поле не должно быть пустым.")


def ask_optional_text(label):
    """
    Спрашивает текст, который можно оставить пустым.
    Возвращает введённую строку или "-", если пользователь нажал Enter.
    """
    text = input(f"{label}: ").strip()
    return text if text else "-"


def ask_time(label):
    """Спрашивает время в формате ЧЧ:ММ и возвращает строку в том же формате."""
    while True:
        raw = input(f"{label} (ЧЧ:ММ): ").strip()
        try:
            dt = datetime.strptime(raw, "%H:%M")
            return dt.strftime("%H:%M")
        except ValueError:
            print("  ! Неверный формат времени. Пример: 08:30")


def ask_optional_time(label):
    """
    Спрашивает время в формате ЧЧ:ММ.
    Пустой ввод (Enter) означает «не было» — возвращаем пустую строку.
    Используется для прибытия спецслужб.
    """
    while True:
        raw = input(f"{label} (ЧЧ:ММ или Enter, если не было): ").strip()

        # ПЕРВЫМ ДЕЛОМ проверяем пустой ввод — это и есть «не было»
        if raw == "":
            return ""

        # Только теперь пытаемся распарсить время
        try:
            dt = datetime.strptime(raw, "%H:%M")
            return dt.strftime("%H:%M")
        except ValueError:
            print("  ! Неверный формат времени. Пример: 08:30 (или Enter)")


def ask_rate():
    """
    Спрашивает ставку за смену в рублях.
    Пустой ввод — ставка 0.
    Возвращает число (int или float).
    """
    while True:
        raw = input("Ставка за смену (руб., можно оставить пустым): ").strip()
        if raw == "":
            return 0
        # Разрешаем и точку, и запятую как десятичный разделитель
        raw = raw.replace(",", ".")
        try:
            value = float(raw)
            if value < 0:
                print("  ! Ставка не может быть отрицательной.")
                continue
            # Если число целое — вернём int, чтобы в CSV было красивее
            return int(value) if value.is_integer() else round(value, 2)
        except ValueError:
            print("  ! Введите число, например: 2500 (или Enter)")


def calc_hours(start_str, end_str):
    """
    Считает длительность смены в часах.
    Смены суточные, поэтому если конец <= начала — считаем,
    что смена перешла через полночь, и прибавляем 24 часа.
    Пример: 08:00 -> 08:00 = 24 часа (следующий день).
    """
    fmt = "%H:%M"
    start = datetime.strptime(start_str, fmt)
    end = datetime.strptime(end_str, fmt)

    # Разница в часах
    diff = (end - start).total_seconds() / 3600

    # Если разница <= 0 — значит, конец наступил на следующий день
    if diff <= 0:
        diff += 24

    return round(diff, 2)


def ask_event():
    """Спрашивает текст о том, что произошло на смене."""
    while True:
        text = input("Что произошло (текст): ").strip()
        if text:
            return text
        print("  ! Поле не должно быть пустым.")


def ask_services():
    """
    Двухэтапный опрос спецслужб:

    Этап 1. По очереди спрашиваем прибытие для каждой службы.
            Пустой ввод = не приезжал.
    Этап 2. Для тех, кто прибыл, спрашиваем убытие.

    Возвращает словарь: {имя: (прибытие, убытие)}.
    Пустые значения означают «не было».
    """
    arrivals = {}

    # ЭТАП 1: только прибытия
    print("\n--- Прибытия ---")
    for name in SERVICES:
        arrival = ask_optional_time(f"Прибытие {name}")
        arrivals[name] = arrival
        if arrival == "":
            print(f"  — {name}: не было")

    # ЭТАП 2: убытия только для тех, кто прибыл
    departures = {}
    print("\n--- Убытия ---")
    any_arrivals = False
    for name in SERVICES:
        arrival = arrivals[name]
        if arrival == "":
            # Не приезжал — убытие не спрашиваем
            departures[name] = ""
            continue
        any_arrivals = True
        # Спрашиваем убытие (тут уже не разрешаем пустой ввод — надо время)
        while True:
            raw = input(
                f"{name} прибыл в {arrival}. Время убытия {name} (ЧЧ:ММ): "
            ).strip()
            try:
                dt = datetime.strptime(raw, "%H:%M")
                departures[name] = dt.strftime("%H:%M")
                break
            except ValueError:
                print("  ! Неверный формат времени. Пример: 09:00")

    if not any_arrivals:
        print("  — никто не приезжал")

    return {name: (arrivals[name], departures[name]) for name in SERVICES}


def build_service_cell(arrival, departure):
    """
    Формирует значение для CSV-столбца службы.
    - Если не был: "-"
    - Если прибыл и убыл: "06:35/09:00"
    - Если прибыл, но убытие почему-то пустое: "06:35/-"
    """
    if arrival == "" and departure == "":
        return "-"
    return f"{arrival or '-'}/{departure or '-'}"


def append_record(record):
    """Дописывает одну строку в CSV-файл."""
    with open(LOG_FILE, mode="a", encoding="utf-8", newline="") as f:
        writer = csv.writer(f, delimiter=";")
        writer.writerow(record)


def main():
    print("=== Учёт суточных смен сторожа ===")
    ensure_file_exists()

    # Общие поля смены
    date_str = ask_date()
    fio = ask_required_text("ФИО дежурного сторожа")
    start_str = ask_time("Время начала")
    end_str = ask_time("Время конца")
    hours = calc_hours(start_str, end_str)
    event = ask_event()

    # Ставка и поручения
    rate = ask_rate()
    notes = ask_optional_text("Поручения/заметки на смену (текст, можно оставить пустым)")

    # Спецслужбы: прибытия, затем убытия
    services = ask_services()

    # Формируем значения для столбцов служб
    service_cells = [
        build_service_cell(services[name][0], services[name][1])
        for name in SERVICES
    ]

    # Собираем строку в порядке HEADERS
    record = [
        date_str,
        fio,
        start_str,
        end_str,
        hours,
        event,
        rate,
        notes,
    ] + service_cells
    append_record(record)

    # Итоговый отчёт на экран
    print("\nЗапись сохранена:")
    print(f"  Дата:         {date_str}")
    print(f"  ФИО:          {fio}")
    print(f"  Начало:       {start_str}")
    print(f"  Конец:        {end_str}")
    print(f"  Длительность: {hours} ч")
    print(f"  Событие:      {event}")
    print(f"  Ставка:       {rate} руб.")
    print(f"  Поручения:    {notes}")
    for name, cell in zip(SERVICES, service_cells):
        if cell == "-":
            print(f"  {name}: не было")
        else:
            print(f"  {name}: {cell}")
    print(f"\nФайл: {os.path.abspath(LOG_FILE)}")


if __name__ == "__main__":
    main()

### Что изменилось

**1. ФИО дежурного.**
Новая функция `ask_required_text` — спрашивает и не пропускает пустой ввод. Поле добавлено вторым столбцом в CSV.

**2. Хронология спецслужб (два этапа).**
Функция `ask_services` теперь работает в два прохода:
- **Этап 1 — прибытия:** по очереди спрашиваем «Прибытие <имя>». Enter = не приезжал.
- **Этап 2 — убытия:** только для тех, кто прибыл. Приглашение содержит напоминание: `Ветврач прибыл в 06:35. Время убытия Ветврача (ЧЧ:ММ):`. Убытие обязательно (пустой ввод не принимается), иначе в журнале будет «прибыл, но не убыл».

**3. Заметки и поручения.**
Функция `ask_optional_text`: пустой ввод → сохраняется прочерк `-`.

**4. Зарплата.**
Функция `ask_rate`: пустой ввод → `0`; принимает и точку, и запятую как десятичный разделитель; отрицательные значения отклоняются. Целые числа сохраняются как `int` (без `.0`).

**5. CSV.**
Новая шапка:
```
Дата;ФИО;Время начала;Время конца;Длительность (ч);Событие;Ставка (руб.);Поручения;Ветврач (Приб/Уб);Молокоцех (Приб/Уб);Бухгалтерия (Приб/Уб);Электрик (Приб/Уб)
```
Значения служб формируются функцией `build_service_cell`:
- не был → `-`
- был → `06:35/09:00`
