Автоматически создает чекпойнт сессии по шаблону Протокола.
"""

import os
from datetime import datetime

# ==================== КОНФИГУРАЦИЯ ====================
ARCHIVE_ROOT = "~/ERROR451_ARCHIVE"  # Основная папка архива (измени на свой путь)
# ======================================================

def get_session_data():
    """Запрашивает у пользователя данные для новой сессии."""
    print("\n" + "="*50)
    print("ERROR451 CHECKPOINT MAKER")
    print("="*50)

    session_data = {}
    session_data['number'] = input("Номер сессии (например, #18): ").strip()
    session_data['title'] = input("Краткое название сессии: ").strip()
    session_data['date'] = input("Дата (DD.MM.YYYY, Enter для сегодня): ").strip()
    if not session_data['date']:
        session_data['date'] = datetime.now().strftime("%d.%m.%Y")
    session_data['participants'] = input("Участники (через запятую): ").strip()
    session_data['log_link'] = input("Ссылка на полный лог: ").strip()
    session_data['goal'] = input("Цель сессии: ").strip()
    
    print("\n--- Краткий нарратив ---")
    print("(Введите построчно. Для завершения введите 'END' с новой строки)")
    narrative_lines = []
    while True:
        line = input()
        if line.strip().upper() == 'END':
            break
        narrative_lines.append(line)
    session_data['narrative'] = "\n".join(narrative_lines)

    print("\n--- Ключевые решения и выводы ---")
    decisions_lines = []
    print("(Вводите по одному пункту. Для завершения введите 'END')")
    while True:
        line = input()
        if line.strip().upper() == 'END':
            break
        decisions_lines.append(f"*   {line}")
    session_data['decisions'] = "\n".join(decisions_lines)

    session_data['code'] = input("\nЗначимый код/команды (если есть): ").strip()

    return session_data

def create_checkpoint_file(session_data):
    """Создает и заполняет файл чекпойнта."""
    # Создаем имя файла: checkpoint_sXX.md
    session_num = session_data['number'].replace('#', '')
    filename = f"checkpoint_s{session_num.zfill(2)}.md"
    
    # Формируем содержимое в формате Markdown
    content = f"""# 🧠 Сессия {session_data['number']}: {session_data['title']}

**[АКТИВНАЯ СЕССИЯ - ЗАВЕРШЕНА {session_data['date']}]**
**Дата:** {session_data['date']}
**Участники:** {session_data['participants']}
**Ссылка на полный лог:** [{session_data['log_link']}]({session_data['log_link']})

### 🎯 Цель сессии
{session_data['goal']}

### 📝 Краткий нарратив
{session_data['narrative']}

### ✅ Ключевые решения и выводы
{session_data['decisions']}

### 💻 Код и команды
```bash
{session_data['code'] if session_data['code'] else "# Команды не требовались"}
```
---
*Чекпойнт создан автоматически инструментом ERROR451 Checkpoint Maker v0.1*
"""
    
    # Определяем путь для сохранения
    save_dir = os.path.expanduser(os.path.join(ARCHIVE_ROOT, "checkpoints"))
    os.makedirs(save_dir, exist_ok=True)
    filepath = os.path.join(save_dir, filename)
    
    # Записываем файл
    with open(filepath, 'w', encoding='utf-8') as f:
        f.write(content)
    
    return filepath

def main():
    """Основная функция."""
    print("Инициализация системы ERROR451...")
    
    # Получаем данные
    data = get_session_data()
    
    # Создаем файл
    print("\nСоздание чекпойнта...")
    saved_path = create_checkpoint_file(data)
    
    # Результат
    print("\n" + "="*50)
    print(f"✅ ЧЕКПОЙНТ УСПЕШНО СОЗДАН!")
    print(f"Файл: {saved_path}")
    print("="*50)
    
    # Предложение сразу открыть файл
    open_file = input("\nОткрыть файл для просмотра? (y/N): ").strip().lower()
    if open_file == 'y':
        os.startfile(saved_path) if os.name == 'nt' else os.system(f'open "{saved_path}"')

if __name__ == "__main__":
    main()
```

### 🔒 **В чём разница и чем грозит HTTP**

*   **HTTP** — обычное соединение. Все данные (пароли, данные карт, переписка) передаются как открытый текст. Их может перехватить кто угодно в сети.
*   **HTTPS** — защищённое соединение. Данные шифруются. Пользователи видят **замочек 🔒** в адресной строке — знак доверия.

**Риски работы по HTTP:**
1.  **Безопасность:** Перехват данных, подмена страниц (фишинг).
2.  **Доверие:** Современные браузеры помечают HTTP-сайты как **«Не защищено»**. Для бизнес-сайта это профанация, клиенты будут уходить.
3.  **SEO:** Google и Яндекс **понижают** HTTP-сайты в поиске. Вы будете терять клиентов, даже не зная об этом.
4.  **Функции:** Многие современные функции браузера (геолокация, push-уведомления) работают **только** по HTTPS.
